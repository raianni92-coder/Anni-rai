#include <stdio.h>
int main()
{
    char a[50];
    int i=0,j=0,vowel=0;
    printf("enter the string:");
    fgets(a,sizeof(a),stdin);
    while(a[i]!='\0'&&a[i]!='\n')
    {
        j=j+1;
        i=i+1;
    }
    printf("length of the string=%d",j);
   
    for(i=0;i<j;i++)
    {
       if(a[i]=='a'||a[i]=='e'||a[i]=='i'||a[i]=='o'||a[i]=='u')
       {
           vowel=vowel+1;
       }
    }   
    printf("\nnumber of vowels=%d",vowel);
    return 0;
}
