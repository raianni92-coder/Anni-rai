#include <stdio.h>
int main()
{
    int a[4][4],row,column,i,j,k,b;
    printf("enter no of rows:");
    scanf("%d",&row);
    printf("enter no of columns:");
    scanf("%d",&column);
    printf("enter matrix elements");
    for(i=0;i<row;i++)
    {
        for(j=0;j<column;j++)
        {
            scanf("%d",&a[i][j]);
        }
    }
    printf("enter number to find frequency:");
    scanf("%d",&k);
    b=0;
    for (i=0;i<row;i++)
    {
        for(j=0;j<column;j++)
        {
           if(k==a[i][j])
           {
               b=b+1;
           }
        }
        
    }
    printf("frequency of the given number is:%d",b);
   
    return 0;
    
}
