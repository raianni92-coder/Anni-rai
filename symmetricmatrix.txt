{
    int a[4][4],b[4][4],row,column,i,j,k=1;

    scanf("%d",&column);
    printf("enter matrix elements");
    for(i=0;i<row;i++)
    {
        for(j=0;j<column;j++)
        {
            scanf("%d",&a[i][j]);
        }
    }
    for(i=0;i<row;i++)

        for(j=0;j<column;j++)
        {
           if( a[j][i]!=a[i][j])

               k=0;
           }
        }
    }
