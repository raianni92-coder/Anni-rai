#include <stdio.h>
#include<math.h>
int main()
{
    int a[100],n,i,k,b=0;
    printf("enter size of an array:");
    scanf("%d",&n);
    for (i=0;i<n;i++)
    {
        printf("enter array element:");
        scanf("%d",&a[i]);
    }
    printf("enter elements to be searched:");
    scanf("%d",&k);
    for(i=0;i<n;i++)
    {
        if(a[i]==k)
        {
            b=1;
            break;
        }
    }
    if(b==1)
    {
        printf("element found");
    }
    else
    {
        printf("element not found");
    }
    
    return 0;
}
