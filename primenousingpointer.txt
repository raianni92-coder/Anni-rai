#include <stdio.h>

int main()
{
    int n, i, *p, count = 0;

    printf("Enter number: ");
    scanf("%d", &n);

    p = &n;   
    for(i = 1; i <= *p; i++)
    {
        if(*p % i == 0)
            count++;
    }

    if(count == 2)
        printf("Prime number");
    else
        printf("Not prime");

    return 0;
}
