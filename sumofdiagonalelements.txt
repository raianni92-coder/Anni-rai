#include <stdio.h>
int main()
{
    int a[4][4],row,column,i,j,k;
    printf("enter no of rows:");
    scanf("%d",&row);
    printf("enter no of columns:");
    scanf("%d",&column);
    printf("enter matrix elements");
    for(i=0;i<row;i++)
    {
        for(j=0;j<column;j++)
        {
            scanf("%d",&a[i][j]);
        }
    }
   
    for (i=0;i<row;i++)
    {
        k=0;
        for(j=0;j<column;j++)
        {
           if(i==j)
           {
               k=k+a[i][j];
           }
        }
        printf("sum of diagonal elements are:%d",k);
    }
   
    return 0;
    
}
