#include <stdio.h>
int main()
{
    char a[50];
    int i=0,j=0;
    printf("enter the string:");
    fgets(a,sizeof(a),stdin);
    while(a[i]!='\0'&&a[i]!='\n')
    {
        j=j+1;
        i=i+1;
    }
    printf("length of the string=%d",j);
    printf("\nreversed string=");
    for(i=j-1;i>=0;i--)
    {
        printf("%c",a[i]);
    }
    return 0;
}
