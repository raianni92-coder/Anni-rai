#include <stdio.h>
int main()
{
    int a[4][4],row,column,i,j,k,b;
    printf("enter no of rows:");
    scanf("%d",&row);
    printf("enter no of columns:");
    scanf("%d",&column);
    printf("enter matrix elements");
    for(i=0;i<row;i++)
    {
        for(j=0;j<column;j++)
        {
            scanf("%d",&a[i][j]);
        }
    }
    printf("sum of rows;\n");
    for (i=0;i<row;i++)
    {
        k=0;
        for(j=0;j<column;j++)
        {
            k=k+a[i][j];
            
        }
        printf("sum of %d row is %d\n",i+1,k);
    }
    for (j=0;j<column;j++)
    {
        b=0;
        for(i=0;i<row;i++)
        {
            b=b+a[i][j];
        }
        printf("sum of%d column is %d\n",j+1,b);
    }
    return 0;
    
}
