#include <stdio.h>
#include<math.h>
int main()
{
    int a,b,i,gcd,lcm;
    printf("enter two numbers:");
    scanf("%d%d", &a,&b);
    for (i=1;i<=a && i<=b; i++)
    {
        if(a%i==0&&b%i==0)
        gcd=i;
    }
    lcm=(a*b)/gcd;
    printf("gcd=%d\nlcm=%d",gcd,lcm);
    return 0;
}
