#include <stdio.h>
#include<math.h>
int main()
{
    int a;
    printf("enter the year");
    scanf("%d", &a);
    if (a%4==0)
    {
        if(a%400==0)
        {
            printf("the year is a leap year");
        }
        else if(a%100==0)
        {
            printf("the year is not a leap year");
        }
    }
    else
    {
        printf("not a leap year");
    }
    return 0;
}
