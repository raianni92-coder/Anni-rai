#include <stdio.h>
#include<math.h>
int main()
{
    float a[5],sum=0,avg;
    int i;
    for(i=0;i<5;i++)
    {
        printf("enter array element:");
        scanf("%f",&a[i]);
    }
    for(i=0;i<5;i++)
    {
        sum=sum+a[i];
    }
    printf("sum of array=%f",sum);
    avg=sum/5;
    printf("average of array=%f",avg);
    return 0;
}
