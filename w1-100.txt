#include <stdio.h>
int main() {
    int i, num, rem, sum;

    for(i = 1; i <= 100; i++) {
        num = i;
        sum = 0;

        while(num != 0) {
            rem = num % 10;
            sum += rem * rem * rem;
            num /= 10;
        }

        if(sum == i)
            printf("%d ", i);
    }

    return 0;
}
